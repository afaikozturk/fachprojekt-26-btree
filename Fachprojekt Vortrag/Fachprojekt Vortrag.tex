	\begin{frame}{Erkenntnis: Insert skaliert schlechter als Lookup}
		\begin{itemize}
			\item Insert braucht Write-Locks, Lookup nur Read-Locks
			\item Stall-Anteil (\texttt{stalls / cycles}) steigt bei Insert deutlich
			stärker als bei Lookup -- TODO: konkrete Prozentwerte einfügen
			\item Erklärung: Lock-Contention am Wurzelbereich beim Baumaufbau,
			\texttt{goto restart}-Zyklen bei konkurrierenden Schreibzugriffen
		\end{itemize}	
	\end{frame}
	
	\begin{frame}{Interessanter Nebenfund}
		\textbf{Uni-Rechner vs. Laptop}\\
